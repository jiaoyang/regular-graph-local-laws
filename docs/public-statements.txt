def SpectralConsequences.PaperRigid : {N d : ℕ} → 3 ≤ d → ℝ → LocalLaw.RegularGraph N d → Prop :=
fun {N d} hd α g =>
  ∀ (i : ℕ) (hi : 2 ≤ i) (hiN : i ≤ N),
    let j := ⟨i - 1, ⋯⟩;
    |SpectralConsequences.sortedEigenvalues ⋯ j - SpectralConsequences.classicalLocation d hd j| ≤ ↑N ^ (-α)
def SpectralConsequences.PaperAllEigenvectors : {N d : ℕ} → LocalLaw.RegularGraph N d → Prop :=
fun {N d} g => ∀ (eig : ℂ) (v : Fin N → ℂ), g.H.mulVec v = eig • v → ‖v‖ ≤ Real.log ↑N ^ 151 / √↑N * √(∑ j, ‖v j‖ ^ 2)
def SpectralConsequences.PaperStructural : {N d : ℕ} → ℝ → ℕ → LocalLaw.RegularGraph N d → Prop :=
fun {N d} c ω g =>
  (∀ (o : Fin N),
      LocalLaw.graphExcess (SpectralConsequences.paperBall g.graph o (SpectralConsequences.paperRadius d N c)) ≤ ω) ∧
    ↑{o | ¬(SpectralConsequences.paperBall g.graph o (SpectralConsequences.paperRadius d N c)).IsTree}.card ≤ ↑N ^ c
def SpectralConsequences.PaperReferenceThreshold : (d : ℕ) → ℕ → 1 ≤ d → Prop :=
fun d ω hd =>
  ∃ C,
    1 ≤ C ∧
      ∀ (V : Type) [inst : DecidableEq V] (g : SpectralConsequences.AllCoreGraph V d),
        g.graph.Connected →
          g.PaperComplexity ω →
            ∀ (z : ℂ),
              0 < z.im →
                (∀ (i j : V),
                    ‖g.green hd z i j‖ ≤
                      C * ‖-LocalLaw.semicircleRoot z * LocalLaw.treeEdgeWeight d‖ ^ g.graph.dist i j) ∧
                  (∀ (i j : V),
                      |(g.green hd z i j).im| ≤
                        C * (LocalLaw.kestenMcKayTransform d (LocalLaw.semicircleRoot z) z).im) ∧
                    ∀ (i : V), ‖LocalLaw.kestenMcKayTransform d (LocalLaw.semicircleRoot z) z‖ / C ≤ ‖g.green hd z i i‖
